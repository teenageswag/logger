cmake_minimum_required(VERSION 3.25)
project(logger LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(logger INTERFACE)
target_include_directories(logger INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/logger/src")
target_compile_features(logger INTERFACE cxx_std_23)
option(LOGGER_SANITIZE "Enable MSVC AddressSanitizer" OFF)
if(MSVC)
  target_compile_options(logger INTERFACE /W4 /permissive- /utf-8)
  if(LOGGER_SANITIZE)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    target_compile_options(logger INTERFACE /fsanitize=address /Zi)
    target_link_options(logger INTERFACE /INCREMENTAL:NO /DEBUG)
  endif()
endif()
option(LOGGER_BUILD_EXAMPLE "Build the interactive example" ON)
if(LOGGER_BUILD_EXAMPLE)
  add_executable(logger_example logger/src/example.cpp)
  target_link_libraries(logger_example PRIVATE logger)
endif()
include(CTest)
option(LOGGER_BUILD_BENCHMARKS "Build the standalone benchmark" OFF)
if(LOGGER_BUILD_BENCHMARKS)
  add_executable(logger_benchmark tests/benchmark.cpp)
  target_link_libraries(logger_benchmark PRIVATE logger)
endif()
if(BUILD_TESTING)
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/logger/src")
  check_cxx_source_compiles("#include <logger/logger.h>\nint main() { log::info(\"{}\", 42); }"
    LOGGER_VALID_FORMAT_COMPILES)
  check_cxx_source_compiles("#include <logger/logger.h>\nint main() { log::info(\"{:d}\", \"text\"); }"
    LOGGER_INVALID_FORMAT_COMPILES)
  if(NOT LOGGER_VALID_FORMAT_COMPILES OR LOGGER_INVALID_FORMAT_COMPILES)
    message(FATAL_ERROR "Logger format-string compile checks failed")
  endif()
  unset(CMAKE_REQUIRED_INCLUDES)
  add_executable(logger_tests tests/logger_tests.cpp)
  target_link_libraries(logger_tests PRIVATE logger)
  add_executable(logger_multi tests/multi_a.cpp tests/multi_b.cpp tests/multi_main.cpp)
  target_link_libraries(logger_multi PRIVATE logger)
  add_test(NAME logger.header_only COMMAND logger_multi)
  set_tests_properties(logger.header_only PROPERTIES TIMEOUT 30)
  if(WIN32)
    add_library(logger_fixture SHARED tests/dll_fixture.cpp tests/dll_fixture.def)
    target_link_libraries(logger_fixture PRIVATE logger)
    add_executable(logger_dll_host tests/dll_host.cpp)
    target_link_libraries(logger_dll_host PRIVATE logger)
    add_test(NAME logger.dll_lifetime COMMAND logger_dll_host "$<TARGET_FILE:logger_fixture>")
    set_tests_properties(logger.dll_lifetime PROPERTIES TIMEOUT 60)
    find_program(LOGGER_PWSH pwsh)
    if(LOGGER_PWSH)
      add_test(NAME logger.owned_console COMMAND "${LOGGER_PWSH}" -NoProfile -File
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/owned_console.ps1"
        -HostExe "$<TARGET_FILE:logger_dll_host>" -Dll "$<TARGET_FILE:logger_fixture>")
      set_tests_properties(logger.owned_console PROPERTIES TIMEOUT 30)
      add_test(NAME logger.json_parse COMMAND "${LOGGER_PWSH}" -NoProfile -File
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/json_parse.ps1"
        -Path "${CMAKE_CURRENT_BINARY_DIR}/sink-fixture.jsonl")
      set_tests_properties(logger.json_parse PROPERTIES FIXTURES_REQUIRED json_data)
    endif()
  endif()
  foreach(case ring drop_newest resize flush shutdown reentrant formatting limits config timer
               io rotation json console model busy_flush)
    add_test(NAME logger.${case} COMMAND logger_tests ${case})
    set_tests_properties(logger.${case} PROPERTIES TIMEOUT 30)
  endforeach()
  set_tests_properties(logger.json PROPERTIES FIXTURES_SETUP json_data)
endif()
